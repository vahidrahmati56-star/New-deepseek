<script type="module">
import * as THREE from 'three';
import { OrbitControls }      from 'three/addons/controls/OrbitControls.js';
import { RoundedBoxGeometry } from 'three/addons/geometries/RoundedBoxGeometry.js';
import { EffectComposer }     from 'three/addons/postprocessing/EffectComposer.js';
import { RenderPass }         from 'three/addons/postprocessing/RenderPass.js';
import { UnrealBloomPass }    from 'three/addons/postprocessing/UnrealBloomPass.js';
import { OutputPass }         from 'three/addons/postprocessing/OutputPass.js';

/* ============================================================
   0. QUALITY TIERING  (premium look, mobile-safe cost)
   ============================================================ */
const UA        = navigator.userAgent;
const IS_MOBILE = /Android|iPhone|iPad|iPod|Mobile/i.test(UA);
const CORES     = navigator.hardwareConcurrency || 4;

const USE_POST    = !IS_MOBILE;                 // bloom on desktop, direct render on phones
const MAX_DPR     = IS_MOBILE ? 1.75 : 2;
const SHADOW_SIZE = IS_MOBILE ? 1024 : 2048;

/* ============================================================
   1. RENDERER
   ============================================================ */
const canvas = document.getElementById('app');

const renderer = new THREE.WebGLRenderer({
  canvas,
  antialias: !USE_POST,          // MSAA handled by composer RT when post is on
  alpha: false,
  stencil: false,
  powerPreference: 'high-performance'
});
renderer.setPixelRatio(Math.min(window.devicePixelRatio, MAX_DPR));
renderer.setSize(window.innerWidth, window.innerHeight);
renderer.shadowMap.enabled = true;
renderer.shadowMap.type    = THREE.PCFSoftShadowMap;
renderer.toneMapping       = THREE.ACESFilmicToneMapping;
renderer.toneMappingExposure = 1.02;

/* ============================================================
   2. SCENE + CAMERA
   ============================================================ */
const scene  = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(38, window.innerWidth / window.innerHeight, 0.1, 120);
camera.position.set(0, 5.9, 7.6);

const controls = new OrbitControls(camera, canvas);
controls.target.set(0, -0.05, 0);
controls.enableDamping  = true;
controls.dampingFactor  = 0.075;
controls.enablePan      = false;
controls.minDistance    = 6.4;
controls.maxDistance    = 13.5;
controls.minPolarAngle  = 0.42;
controls.maxPolarAngle  = 1.24;
controls.autoRotate     = true;
controls.autoRotateSpeed = 0.5;
controls.addEventListener('start', () => { controls.autoRotate = false; });

/* ============================================================
   3. ENVIRONMENT  (procedural IBL — warm key, cool bounce)
   ============================================================ */
function buildEnvironment(renderer) {
  const c = document.createElement('canvas');
  c.width = 512; c.height = 256;
  const ctx = c.getContext('2d');

  const g = ctx.createLinearGradient(0, 0, 0, 256);
  g.addColorStop(0.00, '#FFF7E6');
  g.addColorStop(0.30, '#FFD9A0');
  g.addColorStop(0.50, '#E8A86B');
  g.addColorStop(0.70, '#55668C');
  g.addColorStop(1.00, '#141A29');
  ctx.fillStyle = g;
  ctx.fillRect(0, 0, 512, 256);

  // Crisp specular "sun" → gives ceramics that long glossy streak
  const sun = ctx.createRadialGradient(150, 58, 0, 150, 58, 96);
  sun.addColorStop(0, 'rgba(255,255,255,1)');
  sun.addColorStop(0.35, 'rgba(255,236,200,0.55)');
  sun.addColorStop(1, 'rgba(255,255,255,0)');
  ctx.fillStyle = sun;
  ctx.beginPath(); ctx.arc(150, 58, 96, 0, Math.PI * 2); ctx.fill();

  // Soft cool bounce from below
  const bounce = ctx.createRadialGradient(380, 210, 0, 380, 210, 130);
  bounce.addColorStop(0, 'rgba(150,190,255,0.35)');
  bounce.addColorStop(1, 'rgba(150,190,255,0)');
  ctx.fillStyle = bounce;
  ctx.beginPath(); ctx.arc(380, 210, 130, 0, Math.PI * 2); ctx.fill();

  const tex = new THREE.CanvasTexture(c);
  tex.mapping    = THREE.EquirectangularReflectionMapping;
  tex.colorSpace = THREE.SRGBColorSpace;

  const pmrem = new THREE.PMREMGenerator(renderer);
  pmrem.compileEquirectangularShader();
  const env = pmrem.fromEquirectangular(tex).texture;
  pmrem.dispose();
  tex.dispose();
  return env;
}
scene.environment = buildEnvironment(renderer);

/* ============================================================
   4. BACKDROP  (gentle vertical gradient dome)
   ============================================================ */
(function buildBackdrop() {
  const c = document.createElement('canvas');
  c.width = 32; c.height = 256;
  const ctx = c.getContext('2d');
  const g = ctx.createLinearGradient(0, 0, 0, 256);
  g.addColorStop(0.00, '#2C3757');
  g.addColorStop(0.38, '#1E2740');
  g.addColorStop(0.68, '#141A2B');
  g.addColorStop(1.00, '#0A0D16');
  ctx.fillStyle = g;
  ctx.fillRect(0, 0, 32, 256);

  // Warm bloom of light behind the board
  const glow = ctx.createRadialGradient(16, 118, 0, 16, 118, 86);
  glow.addColorStop(0, 'rgba(255,196,124,0.30)');
  glow.addColorStop(1, 'rgba(255,196,124,0)');
  ctx.fillStyle = glow;
  ctx.fillRect(0, 0, 32, 256);

  const tex = new THREE.CanvasTexture(c);
  tex.colorSpace = THREE.SRGBColorSpace;

  const dome = new THREE.Mesh(
    new THREE.SphereGeometry(48, 32, 20),
    new THREE.MeshBasicMaterial({ map: tex, side: THREE.BackSide, depthWrite: false })
  );
  scene.add(dome);
})();

/* ============================================================
   5. LIGHTING  (warm golden key, cool fill, warm rim)
   ============================================================ */
const key = new THREE.DirectionalLight(0xFFE2B8, 2.45);
key.position.set(5.5, 9.5, 6.5);
key.castShadow = true;
key.shadow.mapSize.set(SHADOW_SIZE, SHADOW_SIZE);
key.shadow.camera.left   = -4.2;
key.shadow.camera.right  =  4.2;
key.shadow.camera.top    =  4.2;
key.shadow.camera.bottom = -4.2;
key.shadow.camera.near   =  0.5;
key.shadow.camera.far    =  30;
key.shadow.bias          = -0.0004;
key.shadow.normalBias    =  0.022;
scene.add(key);

const fill = new THREE.DirectionalLight(0x8FB6FF, 0.72);
fill.position.set(-6.5, 3.2, 5.0);
scene.add(fill);

const rim = new THREE.DirectionalLight(0xFFC27A, 0.95);
rim.position.set(-2.5, 4.5, -7.5);
scene.add(rim);

/* ============================================================
   6. MATERIAL LIBRARY  (the heart of the art direction)
   ============================================================ */
function ceramic(color, { roughness = 0.30, clearcoat = 1.0, ccRough = 0.10, env = 1.0 } = {}) {
  return new THREE.MeshPhysicalMaterial({
    color,
    roughness,
    metalness: 0.0,
    clearcoat,
    clearcoatRoughness: ccRough,
    envMapIntensity: env,
    sheen: 0.25,
    sheenRoughness: 0.85,
    sheenColor: new THREE.Color(0xffffff)
  });
}

function gold(color = 0xE9BE72, roughness = 0.22) {
  return new THREE.MeshPhysicalMaterial({
    color,
    metalness: 1.0,
    roughness,
    envMapIntensity: 1.35,
    clearcoat: 0.5,
    clearcoatRoughness: 0.12
  });
}

const MAT = {
  pedestal : ceramic(0x2A3350, { roughness: 0.34, ccRough: 0.14, env: 1.15 }),
  board    : ceramic(0xE0B98C, { roughness: 0.32, ccRough: 0.12 }),
  tile     : ceramic(0xFFF3E2, { roughness: 0.26, ccRough: 0.08, env: 1.1 }),
  gold     : gold(0xE9BE72, 0.20),
  goldDeep : gold(0xC99A4E, 0.30)
};

/* ============================================================
   7. GEOMETRY BUILD
   ============================================================ */
const world = new THREE.Group();
scene.add(world);

/* --- 7a. Pedestal ------------------------------------------------------- */
const pedestal = new THREE.Mesh(
  new THREE.CylinderGeometry(3.55, 3.80, 0.55, 72, 1),
  MAT.pedestal
);
pedestal.position.y = -0.725;
pedestal.receiveShadow = true;
world.add(pedestal);

const pedestalRing = new THREE.Mesh(
  new THREE.TorusGeometry(3.60, 0.045, 14, 110),
  MAT.gold
);
pedestalRing.rotation.x = -Math.PI / 2;
pedestalRing.position.y = -0.478;
pedestalRing.castShadow = true;
world.add(pedestalRing);

/* --- 7b. Board ---------------------------------------------------------- */
const board = new THREE.Mesh(
  new RoundedBoxGeometry(4.6, 0.45, 4.6, 5, 0.18),
  MAT.board
);
board.position.y = -0.225;
board.castShadow = true;
board.receiveShadow = true;
world.add(board);

/* --- 7c. Gold inlay frame ---------------------------------------------- */
(function buildInlay() {
  const half = 2.0, th = 0.06, hgt = 0.05, y = 0.006;
  const geoH = new RoundedBoxGeometry(half * 2, hgt, th, 3, 0.022);
  const geoV = new RoundedBoxGeometry(th, hgt, half * 2, 3, 0.022);

  const bars = [
    [geoH,  0, y, -half],
    [geoH,  0, y,  half],
    [geoV, -half, y, 0],
    [geoV,  half, y, 0]
  ];
  for (const [geo, x, yy, z] of bars) {
    const m = new THREE.Mesh(geo, MAT.gold);
    m.position.set(x, yy, z);
    m.castShadow = true;
    m.receiveShadow = true;
    world.add(m);
  }
})();

/* --- 7d. Tile grid ------------------------------------------------------ */
const GRID_SPACING = 0.78;
const TILE_SIZE    = 0.66;
const TILE_H       = 0.07;
const TILE_TOP     = TILE_H;

const tileGeo = new RoundedBoxGeometry(TILE_SIZE, TILE_H, TILE_SIZE, 3, 0.03);

for (let r = 0; r < 5; r++) {
  for (let c = 0; c < 5; c++) {
    const tile = new THREE.Mesh(tileGeo, MAT.tile);
    tile.position.set((c - 2) * GRID_SPACING, TILE_H / 2, (r - 2) * GRID_SPACING);
    tile.receiveShadow = true;   // no cast → avoids shadow acne between tiles
    world.add(tile);
  }
}

/* ============================================================
   8. GEMS  (vertical gradient map = "gentle color gradients")
   ============================================================ */
const PALETTE = {
  '1': 0xFF6B6B,   // coral
  '2': 0x4ECDC4,   // teal
  '3': 0xFFC24D,   // amber
  '4': 0xA78BFA    // violet
};

const gradientCache = new Map();
function gemGradient(hex) {
  if (gradientCache.has(hex)) return gradientCache.get(hex);

  const c = document.createElement('canvas');
  c.width = 32; c.height = 128;
  const ctx = c.getContext('2d');

  const base = new THREE.Color(hex);
  const top  = base.clone().lerp(new THREE.Color(0xffffff), 0.72);
  const bot  = base.clone().multiplyScalar(0.66);

  const g = ctx.createLinearGradient(0, 0, 0, 128);
  g.addColorStop(0.00, '#' + top.getHexString());
  g.addColorStop(0.42, '#' + base.getHexString());
  g.addColorStop(1.00, '#' + bot.getHexString());
  ctx.fillStyle = g;
  ctx.fillRect(0, 0, 32, 128);

  const tex = new THREE.CanvasTexture(c);
  tex.colorSpace = THREE.SRGBColorSpace;
  tex.anisotropy = 4;
  gradientCache.set(hex, tex);
  return tex;
}

const gemMaterialCache = new Map();
function gemMaterial(hex) {
  if (gemMaterialCache.has(hex)) return gemMaterialCache.get(hex);
  const m = new THREE.MeshPhysicalMaterial({
    map: gemGradient(hex),
    color: 0xffffff,
    roughness: 0.085,
    metalness: 0.0,
    clearcoat: 1.0,
    clearcoatRoughness: 0.03,
    envMapIntensity: 1.55,
    emissive: new THREE.Color(hex).multiplyScalar(0.16),
    emissiveIntensity: 1.0
  });
  gemMaterialCache.set(hex, m);
  return m;
}

/* --- Contact-shadow decals (cheap, huge perceived-quality win) ---------- */
const shadowDecalTex = (() => {
  const s = 128;
  const c = document.createElement('canvas');
  c.width = c.height = s;
  const ctx = c.getContext('2d');
  const g = ctx.createRadialGradient(s / 2, s / 2, 0, s / 2, s / 2, s / 2);
  g.addColorStop(0.00, 'rgba(255,255,255,1)');
  g.addColorStop(0.48, 'rgba(255,255,255,0.62)');
  g.addColorStop(0.78, 'rgba(255,255,255,0.14)');
  g.addColorStop(1.00, 'rgba(255,255,255,0)');
  ctx.fillStyle = g;
  ctx.fillRect(0, 0, s, s);
  return new THREE.CanvasTexture(c);
})();

const decalGeo = new THREE.PlaneGeometry(0.86, 0.86);
const decalMat = new THREE.MeshBasicMaterial({
  color: 0x1A1206,
  transparent: true,
  opacity: 0.34,
  alphaMap: shadowDecalTex,
  depthWrite: false,
  toneMapped: true
});

/* --- Gem layout (symmetric, readable, pretty) -------------------------- */
const LAYOUT = [
  '.1.2.',
  '3.4.3',
  '.4.1.',
  '2.3.4',
  '.1.2.'
];

const gemGeo = new THREE.SphereGeometry(0.24, 28, 20);
const gems   = [];

for (let r = 0; r < LAYOUT.length; r++) {
  for (let c = 0; c < LAYOUT[r].length; c++) {
    const ch = LAYOUT[r][c];
    if (ch === '.') continue;

    const x = (c - 2) * GRID_SPACING;
    const z = (r - 2) * GRID_SPACING;

    // Contact shadow
    const decal = new THREE.Mesh(decalGeo, decalMat);
    decal.rotation.x = -Math.PI / 2;
    decal.position.set(x, TILE_TOP + 0.0018, z);
    decal.renderOrder = 1;
    world.add(decal);

    // Gem
    const mesh = new THREE.Mesh(gemGeo, gemMaterial(PALETTE[ch]));
    const baseY = 0.305;
    mesh.position.set(x, baseY, z);
    mesh.scale.set(1, 0.92, 1);
    mesh.castShadow = true;
    mesh.receiveShadow = true;
    world.add(mesh);

    gems.push({
      mesh,
      baseY,
      phase: (r * 5 + c) * 0.7,
      spin: 0.25 + (r + c) * 0.045
    });
  }
}

/* ============================================================
   9. POST-PROCESSING
   ============================================================ */
let composer = null;
let bloom = null;

if (USE_POST) {
  const rt = new THREE.WebGLRenderTarget(1, 1, {
    type: THREE.HalfFloatType,
    samples: 4
  });

  composer = new EffectComposer(renderer, rt);
  composer.addPass(new RenderPass(scene, camera));

  bloom = new UnrealBloomPass(
    new THREE.Vector2(window.innerWidth, window.innerHeight),
    0.34,   // strength
    0.62,   // radius
    0.86    // threshold
  );
  composer.addPass(bloom);
  composer.addPass(new OutputPass());
}

/* ============================================================
   10. RESIZE
   ============================================================ */
function resize() {
  const w = window.innerWidth;
  const h = window.innerHeight;
  const dpr = Math.min(window.devicePixelRatio, MAX_DPR);

  renderer.setPixelRatio(dpr);
  renderer.setSize(w, h);

  camera.aspect = w / h;
  camera.updateProjectionMatrix();

  if (composer) {
    composer.setPixelRatio(dpr);
    composer.setSize(w, h);
    // Half-res bloom → cheaper + softer, still premium
    bloom.setSize(w * dpr * 0.5, h * dpr * 0.5);
  }
}
window.addEventListener('resize', resize);
resize();

/* ============================================================
   11. ANIMATION LOOP
   ============================================================ */
const clock = new THREE.Clock();

// Shadows are static (spheres don't change silhouette) → bake once.
renderer.shadowMap.autoUpdate = false;
renderer.shadowMap.needsUpdate = true;

renderer.setAnimationLoop(() => {
  const dt = Math.min(clock.getDelta(), 0.05);
  const t  = clock.elapsedTime;

  for (const g of gems) {
    g.mesh.rotation.y = t * g.spin;
    g.mesh.position.y = g.baseY + Math.sin(t * 1.15 + g.phase) * 0.014;
  }

  controls.update();

  if (composer) composer.render(dt);
  else          renderer.render(scene, camera);
});
